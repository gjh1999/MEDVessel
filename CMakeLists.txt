cmake_minimum_required(VERSION 3.16)
project(MEDVessel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(medvessel INTERFACE)
target_include_directories(medvessel INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(medvessel INTERFACE cxx_std_17)
if(MSVC)
    target_compile_options(medvessel INTERFACE /W4)
else()
    target_compile_options(medvessel INTERFACE -Wall -Wextra -Wpedantic)
endif()
option(MEDVESSEL_ENABLE_OPENMP "Parallelize independent convolution slices" OFF)
if(MEDVESSEL_ENABLE_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(medvessel INTERFACE OpenMP::OpenMP_CXX)
endif()

add_executable(medvessel_benchmark benchmarks/enhancement.cpp)
target_link_libraries(medvessel_benchmark PRIVATE medvessel)

add_executable(medvessel_demo app/main.cpp)
target_link_libraries(medvessel_demo PRIVATE medvessel)

set(MEDVESSEL_DCMTK_SOURCE_DIR "" CACHE PATH "Optional official DCMTK source checkout")
if(MEDVESSEL_DCMTK_SOURCE_DIR)
    enable_language(C)
    set(DCMTK_MODULES "ofstd;oflog;oficonv;dcmdata" CACHE STRING "Required DCMTK modules" FORCE)
    set(BUILD_APPS OFF CACHE BOOL "" FORCE)
    set(DCMTK_DEFAULT_DICT builtin CACHE STRING "" FORCE)
    set(DCMTK_WITH_DOXYGEN OFF CACHE BOOL "" FORCE)
    set(DCMTK_ENABLE_MANPAGES OFF CACHE BOOL "" FORCE)
    set(DCMTK_OVERWRITE_WIN32_COMPILER_FLAGS OFF CACHE BOOL "" FORCE)
    foreach(feature IN ITEMS TIFF PNG XML ZLIB OPENSSL SNDFILE ICONV WRAP OPENJPEG)
        set(DCMTK_WITH_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory(${MEDVESSEL_DCMTK_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/dcmtk EXCLUDE_FROM_ALL)
    add_library(medvessel_dicom src/io/DicomSeries.cpp)
    target_link_libraries(medvessel_dicom PUBLIC medvessel PRIVATE dcmdata)
endif()

set(MEDVESSEL_NIFTI_SOURCE_DIR "" CACHE PATH "Path to official nifti_clib source checkout")
if(MEDVESSEL_NIFTI_SOURCE_DIR)
    enable_language(C)
    find_package(ZLIB REQUIRED)
    add_library(nifti_format STATIC
        ${MEDVESSEL_NIFTI_SOURCE_DIR}/niftilib/nifti1_io.c
        ${MEDVESSEL_NIFTI_SOURCE_DIR}/znzlib/znzlib.c)
    target_include_directories(nifti_format PUBLIC
        ${MEDVESSEL_NIFTI_SOURCE_DIR}/niftilib ${MEDVESSEL_NIFTI_SOURCE_DIR}/znzlib)
    target_compile_definitions(nifti_format PUBLIC HAVE_ZLIB)
    target_link_libraries(nifti_format PUBLIC ZLIB::ZLIB)
    if(UNIX)
        target_link_libraries(nifti_format PUBLIC m)
    endif()
    add_library(medvessel_io src/io/Nifti.cpp)
    target_link_libraries(medvessel_io PUBLIC medvessel PRIVATE nifti_format)
    add_executable(medvessel_cli app/pipeline.cpp)
    target_link_libraries(medvessel_cli PRIVATE medvessel_io)
    if(TARGET medvessel_dicom)
        target_link_libraries(medvessel_cli PRIVATE medvessel_dicom)
        target_compile_definitions(medvessel_cli PRIVATE MEDVESSEL_HAVE_DICOM)
    endif()
    add_executable(medvessel_phantom app/phantom.cpp)
    target_link_libraries(medvessel_phantom PRIVATE medvessel_io)
    add_executable(medvessel_evaluate app/evaluate.cpp)
    target_link_libraries(medvessel_evaluate PRIVATE medvessel_io)
    add_executable(medvessel_export app/export_volume.cpp)
    target_link_libraries(medvessel_export PRIVATE medvessel_io)
    add_executable(medvessel_prepare_case app/prepare_case.cpp)
    target_link_libraries(medvessel_prepare_case PRIVATE medvessel_io)
endif()

# Narrow-path dependencies (nifti/zlib) and argv must agree on UTF-8 on Windows.
# The manifest sets the process ANSI code page without changing the user's system locale.
if(WIN32)
    enable_language(RC)
    foreach(app_target IN ITEMS medvessel_cli medvessel_phantom medvessel_evaluate medvessel_export medvessel_prepare_case)
        if(TARGET ${app_target})
            target_sources(${app_target} PRIVATE app/windows_utf8.rc)
            target_include_directories(${app_target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/app)
        endif()
    endforeach()
endif()

include(CTest)
option(MEDVESSEL_BUILD_NNUNET_TOOLS "Build the isolated nnU-Net Frangi preparation tool" OFF)
if(MEDVESSEL_BUILD_NNUNET_TOOLS)
    add_subdirectory(experiments/nnunet_frangi)
endif()
option(MEDVESSEL_BUILD_GUI "Build the Qt 6 desktop workbench" OFF)
option(MEDVESSEL_ENABLE_VTK "Enable Qt 3D vessel surface and branch interaction" OFF)
if(MEDVESSEL_BUILD_GUI)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets SvgWidgets Concurrent)
    add_library(medvessel_desktop STATIC
        app/gui/JobConfig.cpp app/gui/JobConfig.hpp
        app/gui/AnalysisJob.cpp app/gui/AnalysisJob.hpp
        app/gui/MainWindow.cpp app/gui/MainWindow.hpp
        app/gui/SliceViewer.cpp app/gui/SliceViewer.hpp)
    set_target_properties(medvessel_desktop PROPERTIES AUTOMOC ON)
    target_include_directories(medvessel_desktop PUBLIC app/gui)
    target_link_libraries(medvessel_desktop PUBLIC medvessel Qt6::Widgets Qt6::SvgWidgets Qt6::Concurrent)
    if(TARGET medvessel_io)
        target_link_libraries(medvessel_desktop PUBLIC medvessel_io)
        target_compile_definitions(medvessel_desktop PUBLIC MEDVESSEL_GUI_NIFTI)
    endif()
    if(TARGET medvessel_dicom)
        target_link_libraries(medvessel_desktop PUBLIC medvessel_dicom)
        target_compile_definitions(medvessel_desktop PRIVATE MEDVESSEL_GUI_DICOM)
    endif()
    target_compile_definitions(medvessel_desktop PRIVATE MEDVESSEL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(medvessel_desktop PUBLIC /utf-8)
    endif()
    add_executable(medvessel_gui WIN32 app/gui/main.cpp)
    if(MEDVESSEL_ENABLE_VTK)
        if(NOT TARGET medvessel_io)
            message(FATAL_ERROR "VTK viewer requires MEDVESSEL_NIFTI_SOURCE_DIR")
        endif()
        find_package(VTK 9.2 REQUIRED COMPONENTS CommonCore CommonDataModel CommonTransforms
            FiltersCore FiltersGeneral RenderingCore RenderingOpenGL2 RenderingAnnotation InteractionStyle InteractionWidgets GUISupportQt)
        target_sources(medvessel_desktop PRIVATE app/gui/ThreeDViewer.cpp app/gui/ThreeDViewer.hpp)
        target_compile_definitions(medvessel_desktop PUBLIC MEDVESSEL_GUI_VTK)
        target_link_libraries(medvessel_desktop PUBLIC ${VTK_LIBRARIES})
        vtk_module_autoinit(TARGETS medvessel_desktop medvessel_gui MODULES ${VTK_LIBRARIES})
    endif()
    target_link_libraries(medvessel_gui PRIVATE medvessel_desktop)
    if(WIN32)
        target_sources(medvessel_gui PRIVATE app/windows_utf8.rc)
        target_include_directories(medvessel_gui PRIVATE app)
    endif()
    if(TARGET medvessel_cli)
        add_dependencies(medvessel_gui medvessel_cli)
    endif()
    if(BUILD_TESTING)
        find_package(Qt6 REQUIRED COMPONENTS Test)
        add_executable(gui_tests tests/gui_tests.cpp)
        set_target_properties(gui_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(gui_tests PRIVATE medvessel_desktop Qt6::Test)
        if(MEDVESSEL_ENABLE_VTK)
            vtk_module_autoinit(TARGETS gui_tests MODULES ${VTK_LIBRARIES})
        endif()
        if(TARGET medvessel_cli)
            add_dependencies(gui_tests medvessel_cli medvessel_phantom)
        endif()
        add_test(NAME gui_workbench COMMAND ${CMAKE_COMMAND}
            -DGUI=$<TARGET_FILE:gui_tests>
            -DLOG=${CMAKE_CURRENT_BINARY_DIR}/gui-test-results.txt
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_gui_test.cmake)
        set_tests_properties(gui_workbench PROPERTIES TIMEOUT 90)
    endif()
endif()
if(BUILD_TESTING)
    set(MEDVESSEL_GOOGLETEST_SOURCE_DIR "" CACHE PATH "Optional GoogleTest source checkout")
    option(MEDVESSEL_USE_SYSTEM_GOOGLETEST "Use an installed GoogleTest built with a compatible compiler" OFF)
    if(MEDVESSEL_GOOGLETEST_SOURCE_DIR)
        set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
        set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
        set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
        add_subdirectory(${MEDVESSEL_GOOGLETEST_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/googletest EXCLUDE_FROM_ALL)
    elseif(MEDVESSEL_USE_SYSTEM_GOOGLETEST)
        find_package(GTest QUIET)
    endif()
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE medvessel)
    add_test(NAME core_tests COMMAND core_tests)
    add_executable(filter_tests tests/filter_tests.cpp)
    target_link_libraries(filter_tests PRIVATE medvessel)
    add_test(NAME filter_tests COMMAND filter_tests)
    add_executable(enhancement_tests tests/enhancement_tests.cpp)
    target_link_libraries(enhancement_tests PRIVATE medvessel)
    add_test(NAME enhancement_tests COMMAND enhancement_tests)
    add_executable(segmentation_tests tests/segmentation_tests.cpp)
    target_link_libraries(segmentation_tests PRIVATE medvessel)
    add_test(NAME segmentation_tests COMMAND segmentation_tests)
    add_executable(distance_tests tests/distance_tests.cpp)
    target_link_libraries(distance_tests PRIVATE medvessel)
    add_test(NAME distance_tests COMMAND distance_tests)
    add_executable(geometry_tests tests/geometry_tests.cpp)
    target_link_libraries(geometry_tests PRIVATE medvessel)
    add_test(NAME geometry_tests COMMAND geometry_tests)
    add_executable(thinning_tests tests/thinning_tests.cpp)
    target_link_libraries(thinning_tests PRIVATE medvessel)
    add_test(NAME thinning_tests COMMAND thinning_tests)
    add_executable(graph_tests tests/graph_tests.cpp)
    target_link_libraries(graph_tests PRIVATE medvessel)
    add_test(NAME graph_tests COMMAND graph_tests)
    add_executable(options_tests tests/options_tests.cpp)
    target_link_libraries(options_tests PRIVATE medvessel)
    add_test(NAME options_tests COMMAND options_tests)
    add_executable(evaluation_tests tests/evaluation_tests.cpp)
    target_link_libraries(evaluation_tests PRIVATE medvessel)
    add_test(NAME evaluation_tests COMMAND evaluation_tests)
    find_package(Python3 3.10 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME archive_prefix_integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_archive_prefix.py)
    endif()
    if(TARGET medvessel_dicom)
        add_executable(dicom_tests tests/dicom_tests.cpp)
        target_link_libraries(dicom_tests PRIVATE medvessel_dicom dcmdata)
        add_test(NAME dicom_tests COMMAND dicom_tests)
    endif()
    if(TARGET medvessel_io)
        add_executable(nifti_tests tests/nifti_tests.cpp)
        target_link_libraries(nifti_tests PRIVATE medvessel_io nifti_format)
        add_test(NAME nifti_tests COMMAND nifti_tests)
        add_test(NAME generate_phantom COMMAND medvessel_phantom ${CMAKE_CURRENT_BINARY_DIR}/integration/input)
        set_tests_properties(generate_phantom PROPERTIES FIXTURES_SETUP phantom)
        add_test(NAME pipeline_phantom COMMAND medvessel_cli
            ${CMAKE_CURRENT_BINARY_DIR}/integration/input/phantom.nii.gz
            ${CMAKE_CURRENT_BINARY_DIR}/integration/output)
        set_tests_properties(pipeline_phantom PROPERTIES FIXTURES_REQUIRED phantom FIXTURES_SETUP pipeline)
        add_executable(pipeline_validation tests/pipeline_validation.cpp)
        target_link_libraries(pipeline_validation PRIVATE medvessel_io)
        add_test(NAME validate_pipeline COMMAND pipeline_validation ${CMAKE_CURRENT_BINARY_DIR}/integration)
        set_tests_properties(validate_pipeline PROPERTIES FIXTURES_REQUIRED pipeline)
        add_test(NAME reject_input_collision COMMAND ${CMAKE_COMMAND}
            -DCLI=$<TARGET_FILE:medvessel_cli>
            -DINPUT=${CMAKE_CURRENT_BINARY_DIR}/integration/output/mask.nii.gz
            -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/integration/output
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/reject_collision.cmake)
        set_tests_properties(reject_input_collision PROPERTIES FIXTURES_REQUIRED pipeline)
        add_executable(scenario_validation tests/scenario_validation.cpp)
        target_link_libraries(scenario_validation PRIVATE medvessel_io)
        foreach(scenario IN ITEMS y curved noisy)
            set(scenario_dir ${CMAKE_CURRENT_BINARY_DIR}/scenarios/${scenario})
            add_test(NAME generate_${scenario} COMMAND medvessel_phantom ${scenario_dir} ${scenario})
            set_tests_properties(generate_${scenario} PROPERTIES FIXTURES_SETUP phantom_${scenario})
            add_test(NAME pipeline_${scenario} COMMAND medvessel_cli ${scenario_dir}/phantom.nii.gz ${scenario_dir}/output)
            set_tests_properties(pipeline_${scenario} PROPERTIES FIXTURES_REQUIRED phantom_${scenario} FIXTURES_SETUP pipeline_${scenario})
            if(scenario STREQUAL "y")
                set(expected_branches 3)
            else()
                set(expected_branches 1)
            endif()
            add_test(NAME validate_${scenario} COMMAND scenario_validation ${scenario_dir} ${expected_branches})
            set_tests_properties(validate_${scenario} PROPERTIES FIXTURES_REQUIRED pipeline_${scenario})
        endforeach()
    endif()
    foreach(test_target IN ITEMS core_tests filter_tests enhancement_tests segmentation_tests distance_tests
            geometry_tests thinning_tests graph_tests options_tests evaluation_tests nifti_tests dicom_tests)
        if(TARGET ${test_target} AND TARGET GTest::gtest_main)
            target_compile_definitions(${test_target} PRIVATE MEDVESSEL_HAVE_GTEST)
            target_link_libraries(${test_target} PRIVATE GTest::gtest_main)
        endif()
    endforeach()
    if(MEDVESSEL_ENABLE_OPENMP)
        get_property(medvessel_tests DIRECTORY PROPERTY TESTS)
        set_tests_properties(${medvessel_tests} PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")
    endif()
endif()
